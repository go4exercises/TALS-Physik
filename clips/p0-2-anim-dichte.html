\(\fa{\rho} \lt \rho_\text{W}\) schwimmt; \(\fa{\rho} \gt \rho_\text{W}\) sinkt.
\(\fc{m} = \fa{\rho}\cdot \fb{V}\)

\(\fc{m} = \fa{\rho}\cdot \fb{V}\)
\({}= \fa{2700\,\mathrm{kg/m^3}}\cdot \fb{0.008\,\mathrm{m^3}} = \fc{21.6\,\mathrm{kg}}\)
