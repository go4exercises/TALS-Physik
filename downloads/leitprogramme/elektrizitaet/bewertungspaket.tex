\section*{3 · Musterlösung und Punkteraster}
\textbf{Allgemein:} Ganze Punkte je Zeile. Ein Fehler wird nur einmal abgezogen: Wer mit einem falschen Zwischenergebnis
richtig weiterrechnet, bekommt die folgenden Zeilen (Folgefehler) — auch wenn das Folgeergebnis unplausibel ist; eine fehlende
Plausibilitätsprobe kostet keinen zusätzlichen Punkt. Gleichwertige Wege und Schreibweisen zählen voll.
In der Spalte P: \textbf{(E)}~= Ergebnispunkt — es zählt das Ergebnis \emph{mit richtiger Einheit}, auch ohne Weg; ohne oder mit
falscher Einheit gibt es ihn nicht. \textbf{(W)}~= Wegpunkt — nur mit nachvollziehbarem Weg (Formel, dann Werte mit Einheiten);
steht in einer Weg-Zeile ein Zahlenwert, braucht auch er die Einheit. \textbf{(B)}~= Begründungspunkt — es zählt die fachliche
Aussage in eigenen Worten, eine Formel ist nicht nötig. «Typische Fehler» nennt die Zeilen mit 0 Punkten und die Punktzahl, die bleibt.

\begin{lpaufgabe}{G1}{Elektronen im Draht}{3 P}
\begin{raster}
Ansatz & 1\,(W) & $|Q| = n \cdot e = 2.0 \cdot 10^{19} \cdot 1.602 \cdot 10^{-19}\;\text{C}$\\
Ladung & 1\E & $|Q| \approx 3.20\;\text{C}$ (auch $3.2\;\text{C}$; $-3.20\;\text{C}$ zählt, wenn das Vorzeichen mit den Elektronen begründet ist)\\
Stromstärke & 1\E & $I = \dfrac{Q}{t} = \dfrac{3.20\;\text{C}}{8\;\text{s}} \approx 0.40\;\text{A}$\\
\end{raster}
\fehler{$Q = n / e$ ($1.25 \cdot 10^{38}$): Ansatz und Ladung 0; $I$ mit diesem Wert folgerichtig ($1.56 \cdot 10^{37}\;\text{A}$) zählt $\to$ 1 von 3\,P.
$I = Q \cdot t$: Stromstärke 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{Akku, $50\;\text{Wh}$ bei $10\;\text{V}$}{4 P}
\begin{raster}
Ansatz & 1\,(W) & $W = U \cdot Q$ (konstante Spannung), also $Q = \dfrac{W}{U} = \dfrac{50\;\text{Wh}}{10\;\text{V}}$\\
Ladung in Ah & 1\E & $Q = 5\;\text{Ah}$\\
Ladung in C & 1\E & $Q = 5\;\text{A} \cdot 3600\;\text{s} = 18\,000\;\text{C}$\\
Versorgungsdauer & 1\E & $t = \dfrac{E}{P} = \dfrac{50\;\text{Wh}}{20\;\text{W}} = 2.5\;\text{h}$\\
\end{raster}
\fehler{$Q = U \cdot W = 500$: Ansatz und Ladung in Ah 0; die Umrechnung des eigenen Werts in C ($\cdot 3600$) zählt; Versorgungsdauer unabhängig $\to$ 2 von 4\,P.
Ladung in C mit $\cdot 60$ statt $\cdot 3600$ ($300\;\text{C}$): Ladung in C 0 $\to$ 3 von 4\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Elektroboiler, $6\;\text{kWh}$ in $2.5\;\text{h}$}{4 P}
\begin{raster}
Leistung & 1\E & $P = \dfrac{E}{t} = \dfrac{6\;\text{kWh}}{2.5\;\text{h}} = 2.4\;\text{kW}$\\
Strom & 1\E & $I = \dfrac{P}{U} = \dfrac{2400\;\text{W}}{230\;\text{V}} \approx 10.4\;\text{A}$\\
Kosten & 1\E & $\text{Kosten} = E \cdot \text{Preis} = 6\;\text{kWh} \cdot 0.27\;\text{CHF/kWh} \approx 1.62\;\text{CHF}$\\
«Stromverbrauch» & 1\B & Strom wird nicht verbraucht, er fliesst zurück. Umgesetzt und bezahlt wird Energie.\\
\end{raster}
\fehler{$P = E \cdot t = 15\;\text{kW}$: Leistung 0; Strom folgerichtig ($65.2\;\text{A}$) zählt $\to$ 3 von 4\,P.
$I = \frac{2.4}{230}$ ohne Umrechnung in W ($0.0104\;\text{A}$): Strom 0 $\to$ 3 von 4\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G4}{Kabel für den Rasenmäher}{4 P}
\begin{raster}
Höchster Widerstand & 1\E & $R = \dfrac{U}{I} = \dfrac{6\;\text{V}}{8\;\text{A}} = 0.75\;\Omega$\\
Leiterlänge & 1\,(W) & hin und zurück: $l = 2 \cdot 30\;\text{m} = 60\;\text{m}$\\
Mindestquerschnitt & 1\E & $A = \dfrac{\rho \cdot l}{R} = \dfrac{0.017\;\Omega\,\text{mm}^2/\text{m} \cdot 60\;\text{m}}{0.75\;\Omega} = 1.36\;\text{mm}^2$\\
Wahl & 1\B & $1.5\;\text{mm}^2$ (der kleinste Querschnitt über $1.36\;\text{mm}^2$; $0.75\;\text{mm}^2$ ist zu dünn). Gewählt nur nach dem Spannungsabfall, keine vollständige Installationsbemessung.\\
\end{raster}
\fehler{Nur $30\;\text{m}$ gerechnet ($A = 0.68\;\text{mm}^2$, Wahl $0.75\;\text{mm}^2$): Leiterlänge 0, Rest folgerichtig $\to$ 3 von 4\,P.
$R = U \cdot I = 48\;\Omega$: Höchster Widerstand 0, Querschnitt folgerichtig ($0.021\;\text{mm}^2$) zählt, Wahl $0.75\;\text{mm}^2$ zählt $\to$ 3 von 4\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{$R_1 = 40\;\Omega$, $R_2$ unbekannt, $12\;\text{V}$, $0.2\;\text{A}$}{5 P}
\begin{raster}
a) $R_\text{ges}$ & 1\E & $R_\text{ges} = \dfrac{U}{I} = \dfrac{12\;\text{V}}{0.2\;\text{A}} = 60\;\Omega$\\
a) $R_2$ & 1\E & $R_2 = R_\text{ges} - R_1 = 60\;\Omega - 40\;\Omega = 20\;\Omega$\\
a) Teilspannungen & 1\E & $U_1 = I \cdot R_1 = 8\;\text{V}$, $U_2 = I \cdot R_2 = 4\;\text{V}$ (beide nötig)\\
b) $R_\text{ges}$ parallel & 1\E & $R_\text{ges} = \dfrac{R_1 \cdot R_2}{R_1 + R_2} = \dfrac{40\;\Omega \cdot 20\;\Omega}{60\;\Omega} \approx 13.3\;\Omega$\\
b) Gesamtstrom & 1\E & $I = \dfrac{U}{R_\text{ges}} = \dfrac{12\;\text{V}}{13.3\;\Omega} \approx 0.90\;\text{A}$ (oder $\dfrac{12\;\text{V}}{40\;\Omega} + \dfrac{12\;\text{V}}{20\;\Omega} = 0.3\;\text{A} + 0.6\;\text{A}$)\\
\end{raster}
\fehler{Kehrwert vergessen ($R_\text{ges} = 0.075\;\Omega$): b)~$R_\text{ges}$ 0; der Gesamtstrom zählt folgerichtig, ob über $\frac{12\;\text{V}}{0.075\;\Omega} = 160\;\text{A}$ oder über $I_1 + I_2 = 0.9\;\text{A}$ $\to$ 4 von 5\,P.
$R_2 = 60\;\Omega$ (Gesamtwiderstand für $R_2$ gehalten): $R_2$ 0; Teilspannungen ($8\;\text{V}$ und $12\;\text{V}$) und b) mit $60\;\Omega$ ($24\;\Omega$, $0.5\;\text{A}$) folgerichtig $\to$ 4 von 5\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Waschmaschine mit Isolationsfehler}{5 P}
\begin{raster}
a) Fehlerstrom & 1\E & $I_\text{F} = \dfrac{U}{R} = \dfrac{230\;\text{V}}{2\;\Omega} = 115\;\text{A}$\\
a) Wer trennt & 1\B & Der LS B13 trennt sofort: $115\;\text{A}$ sind mehr als das Fünffache seiner $13\;\text{A}$, dort löst er magnetisch aus (Kurzschlussauslösung). Der FI spricht ebenfalls an, weil $115\;\text{A}$ auf dem Neutralleiter fehlen. (Einer der beiden genügt für den Punkt, wenn die Begründung stimmt; welcher zuerst öffnet, ist nicht verlangt.)\\
b) Körperstrom & 1\E & $I_\text{K} = \dfrac{230\;\text{V}}{1800\;\Omega} \approx 0.128\;\text{A} = 128\;\text{mA}$\\
b) Wer schützt & 1\B & Der FI: $128\;\text{mA}$ fehlen auf dem Rückweg, mehr als $30\;\text{mA}$ — er muss trennen (beim gut Vierfachen spätestens nach $150\;\text{ms}$; die Zeit ist nicht verlangt).\\
b) Warum nicht der LS & 1\B & $0.128\;\text{A}$ liegen weit unter $13\;\text{A}$: Der Leitungsschutzschalter merkt nichts, er schützt die Leitung, nicht den Menschen.\\
\end{raster}
\fehler{$I_\text{K} = 230\;\text{V} \cdot 1800\;\Omega$: Körperstrom 0; «der FI schützt» zählt, wenn mit Hin- und Rückstrom begründet; «der LS löst aus» gibt die letzte Zeile nicht $\to$ 3 von 5\,P.
In a) nur «der FI trennt» ohne Begründung: a)~Wer trennt 0 $\to$ 4 von 5\,P.}
\end{lpaufgabe}
