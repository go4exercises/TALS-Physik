\section*{3 · Musterlösung und Punkteraster}
\textbf{Allgemein:} Ganze Punkte je Zeile. Ein Fehler wird nur einmal abgezogen: Wer mit einem falschen Zwischenergebnis
richtig weiterrechnet, bekommt die folgenden Zeilen (Folgefehler) — auch wenn das Folgeergebnis unplausibel ist; eine fehlende
Plausibilitätsprobe kostet keinen zusätzlichen Punkt. Gleichwertige Wege und Schreibweisen zählen voll.
In der Spalte P: \textbf{(E)}~= Ergebnispunkt — es zählt das Ergebnis \emph{mit richtiger Einheit}, auch ohne Weg; ohne oder mit
falscher Einheit gibt es ihn nicht. \textbf{(W)}~= Wegpunkt — nur mit nachvollziehbarem Weg (Formel, dann Werte mit Einheiten);
steht in einer Weg-Zeile ein Zahlenwert, braucht auch er die Einheit. \textbf{(B)}~= Begründungspunkt — es zählt die fachliche
Aussage in eigenen Worten, eine Formel ist nicht nötig. «Typische Fehler» nennt die Zeilen mit 0 Punkten und die Punktzahl, die bleibt.

\begin{lpaufgabe}{G1}{Begriffe}{3 P}
\begin{raster}
Beschleunigung & 1\B & Änderung der Geschwindigkeit pro Zeit: $a = \dfrac{\Delta v}{\Delta t}$, Einheit $\text{m/s}^2$ (Worte und Formel nötig)\\
Bahnkurve & 1\B & die Linie, die der Schwerpunkt eines Körpers im Lauf der Zeit durchläuft\\
Uhrzeiger & 1\B & ein Kreis um die Zeigerachse\\
\end{raster}
\fehler{«Beschleunigung ist Geschwindigkeit pro Zeit» ohne «Änderung»: Beschleunigung 0 $\to$ 2 von 3\,P. Bahnkurve ohne Bezug zum Ort des Körpers (etwa «die Strecke»): Bahnkurve 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{Velofahrt: $12\;\text{s}$ mit $6\;\text{m/s}$, dann $3\;\text{s}$ Bremsen}{5 P}
\begin{raster}
a) Beschleunigung & 1\E & $a = \dfrac{\Delta v}{\Delta t} = \dfrac{0 - 6\;\text{m/s}}{3\;\text{s}} = -2\;\text{m/s}^2$ (Vorzeichen nötig)\\
b) Weg gleichförmig & 1\E & $s_1 = v \cdot t = 6\;\text{m/s} \cdot 12\;\text{s} = 72\;\text{m}$\\
b) Bremsweg, Summe & 1\E & $s_2 = \tfrac12 \cdot 6\;\text{m/s} \cdot 3\;\text{s} = 9\;\text{m}$ (Dreiecksfläche oder $v_0 \cdot t + \tfrac12 \cdot a \cdot t^2$); zusammen $s = 81\;\text{m}$\\
c) Durchschnitt & 1\E & $\bar v = \dfrac{\Delta s}{\Delta t} = \dfrac{81\;\text{m}}{15\;\text{s}} = 5.4\;\text{m/s}$\\
d) Skizze & 1\B & Achsen mit $t$ in s und $v$ in m/s; waagrechte Linie bei $6\;\text{m/s}$ bis $12\;\text{s}$, dann eine Gerade auf $0$ bei $15\;\text{s}$\\
\end{raster}
\fehler{$a = +2\;\text{m/s}^2$: a) 0 $\to$ 4 von 5\,P. Bremsweg ohne $\tfrac12$ ($18\;\text{m}$, zusammen $90\;\text{m}$): Bremsweg 0, $\bar v = 6\;\text{m/s}$ folgerichtig $\to$ 4 von 5\,P.
$\bar v = \tfrac{6 + 0}{2}\;\text{m/s} = 3\;\text{m/s}$ (Mittel zweier Geschwindigkeiten): c) 0 $\to$ 4 von 5\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Stein von der $45\;\text{m}$ hohen Klippe, waagrecht $12\;\text{m/s}$}{5 P}
\begin{raster}
Fallzeit, Ansatz & 1\,(W) & senkrecht wie im freien Fall: $h = \tfrac12 \cdot g \cdot t^2$, also $t = \sqrt{\dfrac{2h}{g}}$\\
Fallzeit & 1\E & $t = \sqrt{\dfrac{2 \cdot 45\;\text{m}}{9.81\;\text{m/s}^2}} \approx 3.03\;\text{s}$\\
Auftreffpunkt & 1\E & $x = v_0 \cdot t = 12\;\text{m/s} \cdot 3.029\;\text{s} \approx 36.3\;\text{m}$\\
Senkrechte Geschw. & 1\E & $v_y = g \cdot t = 9.81\;\text{m/s}^2 \cdot 3.029\;\text{s} \approx 29.7\;\text{m/s}$\\
Fallen lassen & 1\B & ebenfalls rund $3.03\;\text{s}$: Die senkrechte Bewegung hängt nur von Höhe und $g$ ab, nicht von der waagrechten.\\
\end{raster}
\fehler{$t = \dfrac{2h}{g} \approx 9.17\;\text{s}$ (Wurzel vergessen): Ansatz und Fallzeit 0; $x \approx 110\;\text{m}$ und $v_y = 90\;\text{m/s}$ folgerichtig $\to$ 3 von 5\,P.
$t = \dfrac{h}{v_0} = 3.75\;\text{s}$: Ansatz und Fallzeit 0; $x = 45\;\text{m}$ und $v_y \approx 36.8\;\text{m/s}$ folgerichtig $\to$ 3 von 5\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G4}{Schuss vom Boden: $14\;\text{m/s}$ unter $50^\circ$}{3 P}
\begin{raster}
Ansatz & 1\,(W) & $s_x = \dfrac{v_0^2 \cdot \sin(2\alpha)}{g}$ — oder $t_F = \dfrac{2 \cdot v_0 \cdot \sin\alpha}{g} \approx 2.19\;\text{s}$ und $s_x = v_0 \cdot \cos\alpha \cdot t_F$\\
Wurfweite & 1\E & $s_x = \dfrac{(14\;\text{m/s})^2 \cdot \sin 100^\circ}{9.81\;\text{m/s}^2} \approx 19.7\;\text{m}$\\
Anderer Winkel & 1\B & $40^\circ$: Winkel, die zusammen $90^\circ$ ergeben, tragen gleich weit ($\sin 100^\circ = \sin 80^\circ$).\\
\end{raster}
\fehler{Rechner im Bogenmass ($s_x \approx -10.1\;\text{m}$) oder $\sin\alpha$ statt $\sin(2\alpha)$ ($s_x \approx 15.3\;\text{m}$): Wurfweite 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{Kajak: $2.5\;\text{m/s}$, Fluss $50\;\text{m}$ breit, Strömung $1.5\;\text{m/s}$}{5 P}
\begin{raster}
a) über Grund & 1\E & Pfeile senkrecht: $|\vec v| = \sqrt{(2.5\;\text{m/s})^2 + (1.5\;\text{m/s})^2} \approx 2.92\;\text{m/s}$\\
a) Querzeit & 1\E & $t = \dfrac{b}{v_S} = \dfrac{50\;\text{m}}{2.5\;\text{m/s}} = 20\;\text{s}$\\
a) Versatz & 1\E & $d = v_F \cdot t = 1.5\;\text{m/s} \cdot 20\;\text{s} = 30\;\text{m}$\\
b) Winkel & 1\,(W) & $v_S \cdot \cos\beta = -v_F$: $\cos\beta = -0.6$, $\beta \approx 127^\circ$ (gleichwertig: rund $37^\circ$ gegen die Strömung geneigt, von der Querrichtung aus gemessen)\\
b) Querzeit & 1\E & Querkomponente $v_S \cdot \sin\beta = 2\;\text{m/s}$, also $t = \dfrac{50\;\text{m}}{2\;\text{m/s}} = 25\;\text{s}$\\
\end{raster}
\fehler{Querzeit mit $|\vec v|$ ($t \approx 17.1\;\text{s}$): a) Querzeit 0, Versatz folgerichtig ($\approx 25.7\;\text{m}$) $\to$ 4 von 5\,P.
$|\vec v| = 2.5\;\text{m/s} + 1.5\;\text{m/s} = 4\;\text{m/s}$: a) über Grund 0 $\to$ 4 von 5\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Salatschleuder: $r = 12\;\text{cm}$, $300$ Umdrehungen pro Minute}{4 P}
\begin{raster}
$f$ und $\omega$ & 1\E & $f = \dfrac{300}{60\;\text{s}} = 5\;\text{Hz}$, $\omega = 2\pi \cdot f \approx 31.4\;\text{rad/s}$ (beide nötig)\\
Bahngeschwindigkeit & 1\E & $v = \omega \cdot r = 31.42\;\text{rad/s} \cdot 0.12\;\text{m} \approx 3.77\;\text{m/s}$\\
Zentripetalbeschl. & 1\E & $a_z = \omega^2 \cdot r = \dfrac{v^2}{r} \approx 118\;\text{m/s}^2$\\
Richtung & 1\B & zur Kreismitte, also zur Drehachse hin\\
\end{raster}
\fehler{$f = 300\;\text{Hz}$ (pro Minute nicht umgerechnet): $f$ und $\omega$ 0; $v$ und $a_z$ folgerichtig $\to$ 3 von 4\,P.
$a_z = \dfrac{v}{r} \approx 31.4\;\text{m/s}^2$: Zentripetalbeschleunigung 0 $\to$ 3 von 4\,P.}
\end{lpaufgabe}
